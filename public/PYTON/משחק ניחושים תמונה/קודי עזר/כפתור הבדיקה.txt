import tkinter as tk
from tkinter import PhotoImage
import os

class DragDropGame:
    def __init__(self, root):
        self.root = root
        self.root.title("Drag and Drop Game")

        # יצירת Frame עבור התמונה הגדולה
        self.main_frame = tk.Frame(root, width=1000, height=600, bg="white")
        self.main_frame.pack(fill=tk.BOTH, expand=True)

        # יצירת Frame עבור התמונות הקטנות
        self.small_images_frame = tk.Frame(self.main_frame, bg="lightgrey")
        self.small_images_frame.place(x=10, y=10)

        # בדוק אם התמונה קיימת
        bg_image_path = os.path.join(os.getcwd(), "animal1.png")
        if os.path.isfile(bg_image_path):
            self.bg_image = PhotoImage(file=bg_image_path)
            self.bg_label = tk.Label(self.main_frame, image=self.bg_image)
            self.bg_label.place(x=500, y=10)
        else:
            print("Background image file does not exist")

        # יצירת Frame כאזור היעד
        self.target_frame = tk.Frame(self.main_frame, width=400, height=300, bg="lightblue", borderwidth=2, relief="solid")
        self.target_frame.place(x=500, y=300)

        # טעינת התמונות הקטנות
        self.small_images = [PhotoImage(file=os.path.join(os.getcwd(), f"animal{i}.png")) for i in range(1, 10)]
        self.small_image_labels = []

        # הצגת התמונות הקטנות
        self.original_positions = []
        for i, img in enumerate(self.small_images):
            label = tk.Label(self.small_images_frame, image=img, bg="lightgrey")
            label.grid(row=i // 3, column=i % 3, padx=10, pady=10)
            label.bind("<Button-1>", self.on_drag_start)
            label.bind("<B1-Motion>", self.on_drag_motion)
            label.bind("<ButtonRelease-1>", self.on_drag_release)
            self.small_image_labels.append(label)
            # שמור את המיקומים כ-tuple
            self.original_positions.append((label.winfo_x(), label.winfo_y()))

        # למעקב אחרי מידע הגרירה
        self.drag_data = {}

        # כפתור לבדוק את התשובות
        check_button = tk.Button(root, text="Check", command=self.check_positions)
        check_button.pack(side=tk.BOTTOM)

    def on_drag_start(self, event):
        widget = event.widget
        self.drag_data[widget] = {'x': event.x, 'y': event.y}

    def on_drag_motion(self, event):
        widget = event.widget
        x = widget.winfo_x() - self.drag_data[widget]['x'] + event.x
        y = widget.winfo_y() - self.drag_data[widget]['y'] + event.y
        widget.place(x=x, y=y)

    def on_drag_release(self, event):
        widget = event.widget
        wx, wy = widget.winfo_x(), widget.winfo_y()
        tx, ty, tw, th = self.target_frame.winfo_x(), self.target_frame.winfo_y(), self.target_frame.winfo_width(), self.target_frame.winfo_height()
        if tx <= wx <= tx + tw and ty <= wy <= ty + th:
            print("Image placed inside the target area")
        else:
            print("Image returned to its original position")
            self.reset_image_position(widget)

    def reset_image_position(self, widget):
        index = self.small_image_labels.index(widget)
        original_x, original_y = self.original_positions[index]
        widget.place(x=original_x, y=original_y)

    def check_positions(self):
        correct_count = 0
        for label in self.small_image_labels:
            x, y = label.winfo_x(), label.winfo_y()
            tx, ty, tw, th = self.target_frame.winfo_x(), self.target_frame.winfo_y(), self.target_frame.winfo_width(), self.target_frame.winfo_height()
            if (tx <= x <= tx + tw and ty <= y <= ty + th):
                correct_count += 1
        print(f"Correctly placed images: {correct_count} / 9")

if __name__ == "__main__":
    root = tk.Tk()
    game = DragDropGame(root)
    root.mainloop()

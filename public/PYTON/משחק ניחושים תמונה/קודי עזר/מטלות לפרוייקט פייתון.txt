from tkinter import *


def create_game():
    old_window.destroy() #חובה להרוס את החלון המקורי לפני פתיחת החלון המשני אחרת יהיה שגיאה
    new_window =Tk() #  חלון משני
    new_window.title("!!! START !!!")  # כותרת החלון
    #התמונה חייבת להיות כאן, כי כשהחלון הראשי נסגר, התוכנה לא מכירה את התמונה
    image_background = PhotoImage(file='main background.png')# רקע אחורי
    canvas = Canvas(new_window, width=1800, height=900)
    canvas.pack(fill=BOTH, expand=YES)
    new_window.config(background="#5cfcff")  # רקע אחורי תכלת
    background = canvas.create_image(0, 0, image=image_background, anchor=NW)
    new_window.mainloop() # חובה כדי שהכל יעבוד טוב



def exit_game(): #יציאה מהמשחק
    old_window.destroy()


old_window = Tk() #החלון הראשי
old_window.geometry("620x480") #גודל החלון
old_window.title("Welcome to the guessing game !")# כותרת החלון
old_window.config(background = "#5cfcff") #רקע אחורי תכלת


#התמונות
# חובה לשים את התמונה במשתנה אחרת התמונה לא תישמר בזיכרון ואז לא תיראה
image_START=PhotoImage(file = 'Starting a game.png')
image_exit=PhotoImage(file = 'exit.png')
image_icon=PhotoImage(file = 'icon.png')
old_window.iconphoto(True, image_icon) # האייקון בכותרת החלון בצד שמאל



#הכפתורים
Button(old_window, command = create_game,image=image_START,relief = RAISED,bd = 10,padx = 20,pady = 20,).pack(padx=20, pady=20)# כפתור התחלה
Button(old_window, command = exit_game,image=image_exit,relief = RAISED,bd = 10,padx = 20,pady = 20,).pack() # כפתור יציאה

old_window.mainloop() # שומר על החלון המקורי פתוח עד ליציאה או התחלה

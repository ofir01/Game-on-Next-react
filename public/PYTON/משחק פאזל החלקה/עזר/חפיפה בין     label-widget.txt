from tkinter import *
from PIL import Image, ImageTk
import random

class ImageItem:
    def __init__(self, image_path, position, label, id):
        self.image_path = image_path
        self.position = position
        self.label = label
        self.id = id

    def update_position(self, new_position):
        self.position = new_position

def on_drag_start(event):
    widget = event.widget
    widget.startX = event.x
    widget.startY = event.y

def on_drag_motion(event):
    widget = event.widget
    x = widget.winfo_x() - widget.startX + event.x
    y = widget.winfo_y() - widget.startY + event.y

    # עדכון המיקום של הווידג'ט
    widget.place(x=x, y=y)

    # בדוק אם ה-Label נוגע ב-widget אחר
    if is_overlapping(widget, other_widget):
        print("The widget is overlapping with the label.")

def is_overlapping(widget1, widget2):
    # קבל את הקואורדינטות של הווידג'טים
    x1, y1, x2, y2 = get_widget_coords(widget1)
    a1, b1, a2, b2 = get_widget_coords(widget2)

    # בדוק אם יש חפיפה
    return not (x2 < a1 or x1 > a2 or y2 < b1 or y1 > b2)

def get_widget_coords(widget):
    # קבל את המיקום והגודל של הווידג'ט
    x = widget.winfo_x()
    y = widget.winfo_y()
    width = widget.winfo_width()
    height = widget.winfo_height()

    return (x, y, x + width, y + height)

# יצירת חלון
root = Tk()
root.geometry("720x630")  # גודל החלון
root.title("Drag and Drop Overlap Example")  # כותרת החלון

# הוספת תמונה כווידג'ט
image = Image.open('assets/image.png')
tk_image = ImageTk.PhotoImage(image.resize((200, 200)))

# יצירת label
label = Label(root, image=tk_image)
label.place(x=50, y=50)  # מיקום התמונה ההתחלתי

# יצירת widget נוסף לבדוק אם יש חפיפה
other_widget = Label(root, text="Other Widget", bg="blue", fg="white")
other_widget.place(x=150, y=150, width=100, height=100)

# קישור לאירועי גרירה
label.bind("<ButtonPress-1>", on_drag_start)  # התחלת גרירה
label.bind("<B1-Motion>", on_drag_motion)      # גרירה

root.mainloop()
